4. 修改后的定位逻辑

新增两个辅助函数（放在既有 _settle 之后，复用现有结算辅助）：

def _handover_option(event):
    """取事件的「内部交接」选项：按 dismiss_teacher 语义定位，不写死选项 id。"""
    return next(option for option in event["options"] if option.get("dismiss_teacher"))


def _teacher_departure_event():
    """按业务语义定位「骨干教师去留」事件，而不是靠事件位置或事件池长度。

    事件池由 (seed + semester) % len(事件池) 轮转选出，且条目会随版本继续追加；
    学期逐 1 递增必然遍历事件池的全部条目，因此这里沿真实结算路径推进学期，
    直到出现携带 dismiss_teacher 选项的事件 —— 该标记是「内部交接（解聘一位教师）」
    的语义锚点，与事件排在第几位、池中共有多少条都无关。
    """
    state = create_game("致远高中", seed=3)
    for _ in range(200):
        state, _ = _settle(state)
        event = state["pending_events"][0]
        if any(option.get("dismiss_teacher") for option in event["options"]):
            return event
        zero_cost = next(option for option in event["options"] if option["cost"] == 0)
        state = resolve_event(state, state["version"], event["id"], zero_cost["id"])
    raise AssertionError("事件池中未找到携带 dismiss_teacher 选项的教师去留事件")

三条测试的写法统一变为（业务断言逐条保留）：

def test_transition_dismisses_a_senior_teacher() -> None:
    event = _teacher_departure_event()
    state = create_game("致远高中", seed=3)
    state["pending_events"] = [event]
    senior_before = sum(1 for t in state["teachers"] if t["tier"] == "senior")
    assert senior_before == 1
    state = resolve_event(state, state["version"], event["id"], _handover_option(event)["id"])
    # 内部交接应解聘一位特级教师
    assert sum(1 for t in state["teachers"] if t["tier"] == "senior") == 0
    assert len(state["teachers"]) == 3
